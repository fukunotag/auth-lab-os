  function attachWindowEvents(el,appId) {
    el.addEventListener('pointerdown',()=>focusWindow(appId));
    el.querySelectorAll('[data-window-action]').forEach(btn=>btn.addEventListener('click',e=>{
      e.stopPropagation();
      const action=btn.dataset.windowAction;
      if(action==='close') closeWindow(appId);
      if(action==='minimize') minimizeWindow(appId);
      if(action==='maximize') maximizeWindow(appId);
    }));
    const bar=el.querySelector('[data-drag-handle]');
    let drag=null;
    bar.addEventListener('pointerdown',e=>{
      if(e.target.closest('button') || windows.get(appId).maximized || window.matchMedia('(max-width: 600px)').matches) return;
      drag={x:e.clientX,y:e.clientY,left:el.offsetLeft,top:el.offsetTop};
      bar.setPointerCapture?.(e.pointerId);
    });
    bar.addEventListener('pointermove',e=>{
      if(!drag) return;
      const left=Math.max(0,Math.min(window.innerWidth-el.offsetWidth,drag.left+e.clientX-drag.x));
      const top=Math.max(0,Math.min(window.innerHeight-100,drag.top+e.clientY-drag.y));
      el.style.left=left+'px';el.style.top=top+'px';
    });
    const end=()=>{drag=null};
    bar.addEventListener('pointerup',end);bar.addEventListener('pointercancel',end);
  }

  function base32Encode(bytes) {
    const alphabet='ABCDEFGHIJKLMNOPQRSTUVWXYZ234567'; let bits=0,value=0,out='';
    for(const b of bytes){value=(value<<8)|b;bits+=8;while(bits>=5){out+=alphabet[(value>>>(bits-5))&31];bits-=5;}}
    if(bits>0) out+=alphabet[(value<<(5-bits))&31];
    return out;
  }
  function base32Decode(s) {
    const alphabet='ABCDEFGHIJKLMNOPQRSTUVWXYZ234567';let bits=0,value=0,out=[];
    for(const ch of s.replace(/=+$/,'').toUpperCase()){
      const n=alphabet.indexOf(ch);if(n<0)continue;
      value=(value<<5)|n;bits+=5;
      if(bits>=8){out.push((value>>>(bits-8))&255);bits-=8;}
    }
    return new Uint8Array(out);
  }
  async function totpCode(secret, timeMs=Date.now()) {
    const keyBytes=base32Decode(secret);
    const counter=Math.floor(timeMs/30000);
    const msg=new Uint8Array(8);
    let n=counter;
    for(let i=7;i>=0;i--){msg[i]=n&255;n=Math.floor(n/256);}
    const key=await crypto.subtle.importKey('raw',keyBytes,{name:'HMAC',hash:'SHA-1'},false,['sign']);
    const mac=new Uint8Array(await crypto.subtle.sign('HMAC',key,msg));
    const offset=mac[mac.length-1]&15;
    const binary=((mac[offset]&127)<<24)|((mac[offset+1]&255)<<16)|((mac[offset+2]&255)<<8)|(mac[offset+3]&255);
    return String(binary%1000000).padStart(6,'0');
  }
  async function refreshOtp(winEl) {
    const saved=safeRead('authlab.demo.otp',null);
    if(!saved?.secret){$('#otpCode',winEl).textContent='------';$('#otpCountdown',winEl).textContent='秘密鍵を生成してください';return;}
    try {
      $('#otpSecret',winEl).value=saved.secret;
      $('#otpCode',winEl).textContent=await totpCode(saved.secret);
      const remaining=30-Math.floor((Date.now()/1000)%30);
      $('#otpCountdown',winEl).textContent=`次の更新まで約${remaining}秒`;
    } catch(e) { setStatus(winEl,'このブラウザでは暗号APIを利用できません。HTTPS環境で開いているか確認してください。','warn'); }
  }
  function cloudCall(method, payload, winEl) {
    const status = winEl.querySelector('[data-role="cloud-status"]');
    const fallback = msg => { if(status){status.textContent=msg;status.className='status warn';} else showToast(msg); };
    if(!window.google?.script?.run) {
      fallback('GitHub Pages版ではGAS同期は未接続です。ローカル保存は利用できます。Google Sheets同期は別途安全な連携経路を設定してください。');
      return;
    }
    if(status){status.textContent='GASと通信中…';status.className='status';}
    const runner=google.script.run
      .withSuccessHandler(result=>{
        if(method==='saveCloudData') {
          if(status){status.textContent='Googleスプレッドシートへ保存しました。'+(result.spreadsheetUrl?' 保存先を開く: '+result.spreadsheetUrl:'');status.className='status good';}
        } else {
          if(result.preferences) safeWrite('authlab.preferences',result.preferences);
          if(Array.isArray(result.customDesigns)) safeWrite('authlab.customDesigns',result.customDesigns);
          applyTheme();
          if(status){status.textContent='GASから設定と設計メモを読み込みました。'+(result.spreadsheetUrl?' 保存先: '+result.spreadsheetUrl:'');status.className='status good';}
          const list=winEl.querySelector('#customList');if(list)list.innerHTML=renderCustomList();
        }
      })
      .withFailureHandler(err=>fallback('GAS通信エラー: '+(err?.message||String(err))));
    if(method==='saveCloudData') runner.saveCloudData(clientId,payload);
    else runner.loadCloudData(clientId);
  }
